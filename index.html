const SPREADSHEET_ID = "1QBjapYumy_zPPKsGkXDc6b5NE-i83qXPt3pv64Ap-_U"; const PARENT_FOLDER_ID = "1kuDMYGhfVvUOe1BxlD9Ji-Yb2-ijSlhT"; function doGet(e) { try { const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getActiveSheet(); const data = sheet.getDataRange().getValues(); if (data.length <= 1) { return responseJSON({ status: "success", projects: [] }); } const projects = []; for (let i = 1; i < data.length; i++) { const row = data[i]; let updates = []; try { updates = row[7] ? JSON.parse(row[7]) : []; } catch (err) { updates = []; } projects.push({ id: String(row[0]), title: String(row[1]), percentage: Number(row[2]), status: String(row[3]), color: String(row[4]), notes: String(row[5]), driveUrl: String(row[6]), updates: updates }); } return responseJSON({ status: "success", projects: projects }); } catch (error) { return responseJSON({ status: "error", message: error.toString() }); } } function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { throw new Error("No se recibieron datos."); } const data = JSON.parse(e.postData.contents); const sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getActiveSheet(); // 1. SINCRONIZAR O RELLENAR HOJA COMPLETA if (data.action === "syncLocal") { const projects = data.projects || []; sheet.clearContents(); sheet.appendRow(["id", "title", "percentage", "status", "color", "notes", "driveUrl", "updates"]); projects.forEach(p => { sheet.appendRow([ p.id, p.title, p.percentage, p.status, p.color, p.notes || "", p.driveUrl || "", JSON.stringify(p.updates || []) ]); }); return responseJSON({ status: "success", message: "Hoja de cálculo actualizada con éxito." }); } // 2. ELIMINAR PROYECTO if (data.action === "deleteProject") { const projectId = String(data.id); const folderUrl = data.folderUrl; if (folderUrl) { const match = folderUrl.match(/[-\w]{25,}/); if (match) { try { DriveApp.getFolderById(match[0]).setTrashed(true); } catch(err) {} } } const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === projectId) { sheet.deleteRow(i + 1); break; } } return responseJSON({ status: "success" }); } // 3. GUARDAR / EDITAR PROYECTO INDIVIDUAL const project = data.project || {}; const title = (project.title || data.activityName || "Nueva Actividad").trim(); let driveUrl = project.driveUrl || ""; if (!driveUrl.trim()) { const parentFolder = DriveApp.getFolderById(PARENT_FOLDER_ID); const newFolder = parentFolder.createFolder(title); newFolder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); driveUrl = newFolder.getUrl(); } const rows = sheet.getDataRange().getValues(); let rowIndex = -1; for (let i = 1; i < rows.length; i++) { if (String(rows[i][0]) === String(project.id)) { rowIndex = i + 1; break; } } const rowData = [ project.id || Date.now(), title, Number(project.percentage) || 0, project.status || "en-tiempo", project.color || "bg-teal-400", project.notes || "", driveUrl, JSON.stringify(project.updates || []) ]; if (rowIndex > 0) { sheet.getRange(rowIndex, 1, 1, rowData.length).setValues([rowData]); } else { sheet.appendRow(rowData); } return responseJSON({ status: "success", driveUrl: driveUrl }); } catch (error) { return responseJSON({ status: "error", message: error.toString() }); } } function responseJSON(json) { return ContentService.createTextOutput(JSON.stringify(json)) .setMimeType(ContentService.MimeType.JSON); }
